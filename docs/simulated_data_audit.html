Purpose

This analysis performs the canonical audit and preprocessing of the simulated genomic dataset before any genomic prediction model is fitted.

The audit evaluates file integrity, dimensional consistency, genotype coding, genetic map structure, simulation scenarios, quantitative trait loci (QTL), phenotypic and true genetic values, realized heritability, parental genotypes, marker frequencies, duplicate observations, and linkage disequilibrium.

No genomic prediction model is fitted in this document.

This final audit version explicitly distinguishes the row-order contract from independently verified sample identity and records long-distance LD behavior separately from short-to-medium distance LD decay.

find_project_root <- function(start = getwd()) {
  path <- normalizePath(start, winslash = "/", mustWork = TRUE)

  repeat {
    if (file.exists(file.path(path, "_workflowr.yml"))) {
      return(path)
    }

    parent <- dirname(path)

    if (identical(parent, path)) {
      stop("Could not locate workflowr project root.")
    }

    path <- parent
  }
}

add_audit <- function(check, status, details) {
  stopifnot(status %in% c("PASS", "WARNING", "BLOCKED"))

  audit_log <<- rbind(
    audit_log,
    data.frame(
      Check = check,
      Status = status,
      Details = as.character(details),
      stringsAsFactors = FALSE
    )
  )

  invisible(NULL)
}

# The original control file contains an internal tab in the C1 key
# "Escal\ta". Normalize whitespace inside keys only; numeric values are
# retained exactly as stored in the source file.
normalize_control_key <- function(x) {
  key <- sub(",.*$", "", x)
  key <- trimws(key)
  key <- gsub("[[:space:]]+", "", key)
  tolower(key)
}

control_keys <- vapply(
  control_ascii,
  normalize_control_key,
  character(1)
)

extract_control_value <- function(block, key) {
  block_keys <- vapply(
    block,
    normalize_control_key,
    character(1)
  )

  target_key <- normalize_control_key(
    paste0(key, ",")
  )

  hit <- which(block_keys == target_key)

  if (length(hit) == 0L) {
    return(NA_real_)
  }

  value_text <- trimws(
    sub("^[^,]*,", "", block[hit[1L]])
  )

  suppressWarnings(
    as.numeric(value_text)
  )
}

Linkage disequilibrium

LD is evaluated using the canonical individual-by-marker 0/1/2 matrix. Within-linkage-group marker pairs are evaluated at fixed map distances. A random sample of pairs from different linkage groups provides an interchromosomal reference.

if (monomorphic_markers == 0L) {
  geno_scaled <- scale(
    geno_012,
    center = TRUE,
    scale = TRUE
  )

  pair_r2 <- function(i, j, z, block_size = 500L) {
    stopifnot(length(i) == length(j))

    result <- numeric(length(i))

    if (length(i) == 0L) {
      return(result)
    }

    block_start <- seq(
      1L,
      length(i),
      by = block_size
    )

    for (start in block_start) {
      end <- min(
        start + block_size - 1L,
        length(i)
      )

      b <- start:end

      r <- colSums(
        z[, i[b], drop = FALSE] *
          z[, j[b], drop = FALSE]
      ) / (nrow(z) - 1)

      result[b] <- r^2
    }

    result
  }

  ld_distances <- c(
    0.5,
    1,
    2.5,
    5,
    10,
    25,
    50,
    100,
    200
  )

  ld_rows <- vector(
    "list",
    length(ld_distances)
  )

  for (d in seq_along(ld_distances)) {
    distance <- ld_distances[d]
    r2_all <- numeric()

    for (lg in sort(unique(genetic_map$LG))) {
      idx <- which(genetic_map$LG == lg)

      idx <- idx[
        order(genetic_map$Position_cM[idx])
      ]

      pos <- genetic_map$Position_cM[idx]

      target <- match(
        pos + distance,
        pos
      )

      keep <- !is.na(target)

      if (any(keep)) {
        i <- idx[keep]
        j <- idx[target[keep]]

        r2_all <- c(
          r2_all,
          pair_r2(
            i,
            j,
            geno_scaled
          )
        )
      }
    }

    ld_rows[[d]] <- data.frame(
      Distance_cM = distance,
      Pairs = length(r2_all),
      Mean_r2 = mean(r2_all),
      Median_r2 = median(r2_all)
    )
  }

  ld_summary <- do.call(
    rbind,
    ld_rows
  )

  set.seed(SEED_GLOBAL + 101L)

  inter_i <- sample(
    seq_len(ncol(geno_012)),
    n_inter_pairs,
    replace = TRUE
  )

    seq_len(ncol(geno_012)),
    n_inter_pairs,
    replace = TRUE
  )

  same_lg <- genetic_map$LG[inter_i] ==
    genetic_map$LG[inter_j]

  while (any(same_lg)) {
    inter_j[same_lg] <- sample(
      seq_len(ncol(geno_012)),
      sum(same_lg),
      replace = TRUE
    )

    same_lg <- genetic_map$LG[inter_i] ==
      genetic_map$LG[inter_j]
  }

  inter_r2 <- pair_r2(
    inter_i,
    inter_j,
    geno_scaled
  )

  interchromosomal_ld <- data.frame(
    Distance_cM = NA_real_,
    Pairs = length(inter_r2),
    Mean_r2 = mean(inter_r2),
    Median_r2 = median(inter_r2)
  )

Interpretation of the simulation scenarios

The six scenarios constitute a complexity gradient, rather than a complete factorial experiment between QTL number and heritability.

The nominal scenarios are:

Direct comparisons of QTL-number complexity while nominal heritability is held constant are therefore naturally defined within the following pairs:

  • C1 versus C2 at nominal \(H^2 = 0.7\);
  • C3 versus C4 at nominal \(H^2 = 0.5\);
  • C5 versus C6 at nominal \(H^2 = 0.3\).

Across all six scenarios simultaneously, QTL number increases while nominal heritability decreases. Consequently, a global trend across C1-C6 must not be interpreted as an isolated causal effect of QTL number or heritability.

The simulator specifies Model = 2 for all scenarios. This simulation architecture is retained as part of the data-generating contract and motivates the use of additive, dominance, and epistatic genomic components in the GBLUP-ADE benchmark.

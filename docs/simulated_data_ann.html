Purpose

This module is the canonical ANN workflow for the simulated genomic-data branch. ANN hyperparameter tuning and final outer-CV fitting are internal stages of one modeling method, parallel to GBLUP in the public pipeline.

The four variants are ANN_0, ANN_L1, ANN_L2, and ANN_L1_L2 (displayed as ANN-L1+L2). They use the same multilayer perceptron and differ only in L1/L2 weight penalties.

The page has two independent execution modes:

  • RUN_ANN_TUNING=1: run or refresh nested inner-CV tuning only;
  • RUN_ANN=1: consume the saved canonical tuning selections, select Best_epoch using only the 800 outer-analysis individuals, refit from scratch on all 800 for that epoch count, and only then predict the untouched 200 outer-assessment individuals.

Using both flags intentionally runs the full ANN workflow. Ordinary builds run neither expensive stage and only render saved outputs.

Tuning contract

Only four quantities are treated as hyperparameters:

  1. hidden-layer architecture;
  2. Adam learning rate;
  3. L1 penalty coefficient;
  4. L2 penalty coefficient.

The following are fixed optimization controls, not tuned quantities:

  • activation: ReLU;
  • optimizer: Adam;
  • dropout: 0;
  • maximum epochs available to an early-stopped fit: 2000;
  • patience: 25;
  • minimum validation improvement: \(10^{-6}\);
  • early-validation fraction: 0.15.

The 2000-epoch value is a convergence ceiling, not a fixed final training length. The saved ANN_TUNING_CANONICAL_V1 metadata still contains the historical field final_train_epochs = 2000 because that value belongs to the completed tuning run’s immutable provenance/configuration identifier; the canonical final-fitting stage does not use that field as a refit rule. Epoch diagnostics are excluded from hyperparameter ranking. After hyperparameters are selected, a new early-stopping fit uses only the outer-analysis sample to select Best_epoch; the final network is then refit from scratch on all 800 outer-analysis individuals for exactly Best_epoch.

Architecture range

The canonical architecture catalog deliberately defines a bounded small-to-moderate neural-network search space, ranging from two hidden neurons to 64-32.

Earlier development experiments explored larger networks, including architectures up to 256-128, and showed that increased capacity could improve inner-validation performance in some tasks. Those larger architectures were not carried forward into the canonical grid. Instead, 64-32 was retained as a prespecified computational and methodological upper bound for the final comparison.

This restriction is therefore part of the estimand of the canonical analysis: the reported comparison evaluates GBLUP-ADE against regularized and unregularized ANN models within this bounded architecture class. It should not be interpreted as demonstrating that GBLUP-ADE outperforms every possible higher-capacity neural-network architecture.

safe_cor <- function(p, o) {
  z <- is.finite(p) & is.finite(o)

  if (
    sum(z) < 3L ||
    sd(p[z]) == 0 ||
    sd(o[z]) == 0
  ) {
    return(NA_real_)
  }

  cor(p[z], o[z])
}

safe_rmse <- function(p, o) {
  z <- is.finite(p) & is.finite(o)

  if (!any(z)) {
    return(NA_real_)
  }

  sqrt(
    mean(
      (p[z] - o[z])^2
    )
  )
}

safe_mae <- function(p, o) {
  z <- is.finite(p) & is.finite(o)

  if (!any(z)) {
    return(NA_real_)
  }

  mean(
    abs(
      p[z] - o[z]
    )
  )
}

make_validation_split <- function(
  x,
  frac,
  seed
) {
  set.seed(seed)

  n_validation <- max(
    1L,
    as.integer(
      round(
        length(x) * frac
      )
    )
  )

  validation <- sort(
    sample(
      x,
      n_validation
    )
  )

  list(
    train = sort(
      setdiff(
        x,
        validation
      )
    ),
    validation = validation
  )
}

fit_x_scaler <- function(x) {
  center <- colMeans(x)
  scale <- apply(
    x,
    2,
    sd
  )

  zero <- !is.finite(scale) |
    scale < 1e-8

  scale[zero] <- 1

  list(
    center = center,
    scale = scale,
    zero = zero
  )
}

scale_x <- function(x, scaler) {
  sweep(
    sweep(
      x,
      2,
      scaler$center,
      "-"
    ),
    2,
    scaler$scale,
    "/"
  )
}

fit_y_scaler <- function(y) {
  scale <- sd(y)

  if (
    !is.finite(scale) ||
    scale < 1e-8
  ) {
    scale <- 1
  }

  list(
    center = mean(y),
    scale = scale
  )
}

scale_y <- function(y, scaler) {
  (
    y - scaler$center
  ) / scaler$scale
}

unscale_y <- function(y, scaler) {
  y * scaler$scale +
    scaler$center
}

  ) {
    self$linears <-
      torch::nn_module_list()

    previous <-
      as.integer(input_dim)

    for (h in hidden_units) {
      self$linears$append(
        torch::nn_linear(
          previous,
          as.integer(h)
        )
      )
      previous <- as.integer(h)
    }

    self$output <-
      torch::nn_linear(
        previous,
        1L
      )
  },

  forward = function(x) {
    if (length(self$linears) > 0L) {
      for (
        i in seq_len(
          length(self$linears)
        )
      ) {
        x <- torch::nnf_relu(
          self$linears[[i]](x)
        )
      }
    }

    self$output(x)
  }
)

weight_penalty <- function(
  model,
  l1,
  l2
) {
  weights <- lapply(
    seq_len(
      length(model$linears)
    ),
    function(i) {
      model$linears[[i]]$weight
    }
  )

  weights[[length(weights) + 1L]] <-
    model$output$weight

  zero <- weights[[1]]$sum() * 0
  penalty_l1 <- zero
  penalty_l2 <- zero

  for (w in weights) {
    penalty_l1 <-
      penalty_l1 +
      w$abs()$sum()

    penalty_l2 <-
      penalty_l2 +
      w$pow(2)$sum()
  }

  l1 * penalty_l1 +
    (l2 / 2) * penalty_l2
}

clone_state <- function(model) {
  state <- model$state_dict()

  copy <- lapply(
    state,
    function(x) {
      x$detach()$clone()
    }
  )

  names(copy) <- names(state)

  copy
}

Inner-CV training

The inner analysis set is divided deterministically into:

  • a fitting subset;
  • an early-validation subset.

The frozen inner assessment set remains untouched until prediction of that candidate. Preprocessing is estimated from the fitting subset only.

Early stopping is an optimization mechanism. Its epoch diagnostics are saved but excluded from the tuning rule.

prepare_bundle <- function(
  fit_ids,
  validation_ids,
  assessment_ids,
  scenario_index
) {
  fit_index <- match(
    fit_ids,
    ids
  )
  validation_index <- match(
    validation_ids,
    ids
  )
  assessment_index <- match(
    assessment_ids,
    ids
  )

  if (
    anyNA(
      c(
        fit_index,
        validation_index,
        assessment_index
      )
    )
  ) {
    stop(
      "ANN ID alignment failed."
    )
  }

  x_scaler <- fit_x_scaler(
    geno[
      fit_index,
      ,
      drop = FALSE
    ]
  )

  y_scaler <- fit_y_scaler(
    phenotype[
      fit_index,
      scenario_index
    ]
  )

  list(
    x_fit = scale_x(
      geno[
        fit_index,
        ,
        drop = FALSE
      ],
      x_scaler
    ),
    x_validation = scale_x(
      geno[
        validation_index,
        ,
        drop = FALSE
      ],
      x_scaler
    ),
    x_assessment = scale_x(
      geno[
        assessment_index,
        ,
        drop = FALSE
      ],
      x_scaler
    ),
    y_fit = scale_y(
      phenotype[
        fit_index,
        scenario_index
      ],
      y_scaler
    ),
    y_validation = scale_y(
      phenotype[
        validation_index,
        scenario_index
      ],
      y_scaler
    ),
    y_assessment = phenotype[
      assessment_index,
      scenario_index
    ],
    y_scaler = y_scaler,
    zero_variance_markers =
      sum(x_scaler$zero)
  )
}

train_inner_candidate <- function(
  bundle,
  architecture,
  learning_rate,
  l1,
  l2,
  seed
) {
  set_ann_seed(seed)

  x_fit <- torch::torch_tensor(
    bundle$x_fit,
    dtype = torch::torch_float()
  )$to(device = ANN_DEVICE)

  y_fit <- torch::torch_tensor(
    matrix(
      bundle$y_fit,
      ncol = 1
    ),
    dtype = torch::torch_float()
  )$to(device = ANN_DEVICE)

  x_validation <- torch::torch_tensor(
    bundle$x_validation,
    dtype = torch::torch_float()
  )$to(device = ANN_DEVICE)

  y_validation <- torch::torch_tensor(
    matrix(
      bundle$y_validation,
      ncol = 1
    ),
    dtype = torch::torch_float()
  )$to(device = ANN_DEVICE)

  x_assessment <- torch::torch_tensor(
    bundle$x_assessment,
    dtype = torch::torch_float()
  )$to(device = ANN_DEVICE)

  model <- TorchGenomicMLP(
    ncol(bundle$x_fit),
    ARCHITECTURES[[architecture]]
  )$to(device = ANN_DEVICE)

  optimizer <- torch::optim_adam(
    params = model$parameters,
    lr = learning_rate,
    weight_decay = 0
  )

  best_validation_loss <- Inf
  checkpoint_epoch <- NA_integer_
  checkpoint_state <- NULL
  stale <- 0L
  last_epoch <- 0L
  stopped_by_patience <- FALSE

  sync_device()
  fit_start <-
    proc.time()[["elapsed"]]

  for (
    epoch in seq_len(MAX_EPOCHS)
  ) {
    last_epoch <- epoch

    model$train()
    optimizer$zero_grad()

    prediction <-
      model(x_fit)

    loss <-
      torch::nnf_mse_loss(
        prediction,
        y_fit
      ) +
      weight_penalty(
        model,
        l1,
        l2
      )

    loss$backward()
    optimizer$step()

    model$eval()

    validation_loss <-
      torch::with_no_grad({
        torch::nnf_mse_loss(
          model(x_validation),
          y_validation
        )
      })

    validation_value <-
      as.numeric(
        validation_loss$to(
          device =
            torch::torch_device("cpu")
        )
      )

    if (
      is.finite(validation_value) &&
      validation_value <
        best_validation_loss -
        MIN_DELTA
    ) {
      best_validation_loss <-
        validation_value

      checkpoint_epoch <-
        epoch

      checkpoint_state <-
        clone_state(model)

      stale <- 0L
    } else {
      stale <- stale + 1L
    }

    if (stale >= PATIENCE) {
      stopped_by_patience <- TRUE
      break
    }
  }

  sync_device()
  fit_seconds <-
    proc.time()[["elapsed"]] -
    fit_start

  if (
    is.null(checkpoint_state) ||
    is.na(checkpoint_epoch)
  ) {
    stop(
      "No finite early-validation checkpoint."
    )
  }

  model$load_state_dict(
    checkpoint_state
  )
  model$eval()

  sync_device()
  prediction_start <-
    proc.time()[["elapsed"]]

  prediction_scaled <-
    torch::with_no_grad({
      as.numeric(
        model(x_assessment)$to(
          device =
            torch::torch_device("cpu")
        )
      )
    })

  sync_device()
  prediction_seconds <-
    proc.time()[["elapsed"]] -
    prediction_start

  list(
    prediction = unscale_y(
      prediction_scaled,
      bundle$y_scaler
    ),
    early_stop_epoch =
      as.integer(checkpoint_epoch),
    last_epoch =
      as.integer(last_epoch),
    stopped_by_patience =
      stopped_by_patience,
    hit_epoch_ceiling =
      last_epoch >= MAX_EPOCHS &&
      !stopped_by_patience,
    fit_seconds = fit_seconds,
    prediction_seconds =
      prediction_seconds,
    zero_variance_markers =
      bundle$zero_variance_markers
  )
}

Candidate evaluation and selection

Candidate ranking is deterministic and does not include epoch information.

The ranking hierarchy is:

  1. lower mean inner-assessment RMSE;
  2. higher mean inner-assessment correlation;
  3. fewer network parameters;
  4. smaller total penalty coefficient;
  5. smaller learning rate;
  6. candidate ID.
evaluate_candidate <- function(
  bundles,
  model_name,
  architecture,
  learning_rate,
  l1,
  l2,
  seed_base,
  candidate_id
) {
  rows <- vector(
    "list",
    4L
  )

  for (k in 1:4) {
    fitted <- tryCatch(
      train_inner_candidate(
        bundles[[k]],
        architecture,
        learning_rate,
        l1,
        l2,
        seed_base + k
      ),
      error = function(e) e
    )

    if (
      inherits(
        fitted,
        "error"
      )
    ) {
      rows[[k]] <- data.frame(
        Model = model_name,
        CandidateID = candidate_id,
        Architecture = architecture,
        Learning_rate = learning_rate,
        L1 = l1,
        L2 = l2,
        InnerFold = k,
        RMSE = NA_real_,
        Correlation = NA_real_,
        MAE = NA_real_,
        EarlyStop_epoch = NA_integer_,
        Last_epoch = NA_integer_,
        Stopped_by_patience = NA,
        Hit_epoch_ceiling = NA,
        Fit_seconds = NA_real_,
        Prediction_seconds = NA_real_,
        Status = "error",
        Error = conditionMessage(fitted),
        stringsAsFactors = FALSE
      )
    } else {
      rows[[k]] <- data.frame(
        Model = model_name,
        CandidateID = candidate_id,
        Architecture = architecture,
        Learning_rate = learning_rate,
        L1 = l1,
        L2 = l2,
        InnerFold = k,
        RMSE = safe_rmse(
          fitted$prediction,
          bundles[[k]]$y_assessment
        ),
        Correlation = safe_cor(
          fitted$prediction,
          bundles[[k]]$y_assessment
        ),
        MAE = safe_mae(
          fitted$prediction,
          bundles[[k]]$y_assessment
        ),
        EarlyStop_epoch =
          fitted$early_stop_epoch,
        Last_epoch =
          fitted$last_epoch,
        Stopped_by_patience =
          fitted$stopped_by_patience,
        Hit_epoch_ceiling =
          fitted$hit_epoch_ceiling,
        Fit_seconds =
          fitted$fit_seconds,
        Prediction_seconds =
          fitted$prediction_seconds,
        Status = "ok",
        Error = NA_character_,
        stringsAsFactors = FALSE
      )
    }
  }

  folds <- do.call(
    rbind,
    rows
  )

  ok <- folds$Status == "ok"

  summary <- data.frame(
    Model = model_name,
    CandidateID = candidate_id,
    Architecture = architecture,
    Learning_rate = learning_rate,
    L1 = l1,
    L2 = l2,
    Inner_folds_ok = sum(ok),
    Mean_RMSE = if (all(ok)) {
      mean(folds$RMSE)
    } else {
      NA_real_
    },
    Mean_Correlation = if (all(ok)) {
      mean(folds$Correlation)
    } else {
      NA_real_
    },
    Mean_MAE = if (all(ok)) {
      mean(folds$MAE)
    } else {
      NA_real_
    },
    Median_EarlyStop_epoch = if (all(ok)) {
      median(folds$EarlyStop_epoch)
    } else {
      NA_real_
    },
    Max_Last_epoch = if (all(ok)) {
      max(folds$Last_epoch)
    } else {
      NA_real_
    },
    N_epoch_ceiling = if (all(ok)) {
      sum(folds$Hit_epoch_ceiling)
    } else {
      NA_integer_
    },
    Status = if (all(ok)) {
      "ok"
    } else {
      "incomplete"
    },
    stringsAsFactors = FALSE
  )

  list(
    summary = summary,
    folds = folds
  )
}

choose_candidate <- function(x) {
  eligible <- x[
    x$Status == "ok" &
      is.finite(x$Mean_RMSE),
    ,
    drop = FALSE
  ]

  if (!nrow(eligible)) {
    stop(
      "No complete ANN candidate is available."
    )
  }

  eligible$Parameters <- vapply(
    eligible$Architecture,
    function(architecture) {
      parameter_count(
        ncol(geno),
        ARCHITECTURES[[architecture]]
      )
    },
    numeric(1)
  )

  eligible$Penalty <-
    eligible$L1 +
    eligible$L2

  eligible <- eligible[
    order(
      eligible$Mean_RMSE,
      -eligible$Mean_Correlation,
      eligible$Parameters,
      eligible$Penalty,
      eligible$Learning_rate,
      eligible$CandidateID
    ),
    ,
    drop = FALSE
  ]

  # Epoch diagnostics are intentionally absent from the ranking above.
  eligible[
    1,
    ,
    drop = FALSE
  ]
}

  tuning_folds <- rbind(
    structure_folds,
    do.call(
      rbind,
      penalty_folds
    )
  )

  for (
    column_name in c(
      "Scenario",
      "Repetition",
      "Fold",
      "SplitID",
      "TaskID"
    )
  ) {
    tuning_summary[[column_name]] <-
      task[[column_name]]

    tuning_folds[[column_name]] <-
      task[[column_name]]
  }

  task_seconds <-
    proc.time()[["elapsed"]] -
    task_start

  checkpoint <- list(
    Status = "ok",
    ConfigID =
      ANN_TUNING_CONFIG_ID,
    TaskID = task$TaskID,
    selected = selected_table,
    tuning_summary = tuning_summary,
    tuning_folds = tuning_folds,
    task_seconds = task_seconds
  )

  saveRDS(
    checkpoint,
    checkpoint_file,
    compress = "gzip"
  )

  task_outputs[[task_index]] <-
    checkpoint

  cat(
    sprintf(
      "[%03d/%03d] %s | %.1f s | arch=%s lr=%g\n",
      task_index,
      nrow(task_table),
      task$TaskID,
      task_seconds,
      selected_architecture,
      selected_learning_rate
    )
  )

  gc()
}

Reproducibility outputs

The canonical tuning module produces:

  • candidate-level inner-CV summaries;
  • fold-level candidate results;
  • selected configuration for each Scenario × outer split × ANN variant;
  • a tuning manifest documenting every prespecified grid and fixed training control.

The final-fitting stage below consumes the selected configuration artifact and the tuning manifest. It does not repeat hyperparameter tuning.

Execution

Routine website build without model fitting:

Rscript -e 'workflowr::wflow_build("analysis/simulated_data_ann.Rmd")'

One-task pilot:

ANN_TUNING_PILOT=1 RUN_ANN_TUNING=1 \
Rscript -e 'workflowr::wflow_build("analysis/simulated_data_ann.Rmd")'

Complete canonical tuning:

RUN_ANN_TUNING=1 \
Rscript -e 'workflowr::wflow_build("analysis/simulated_data_ann.Rmd")'

Completed tasks are checkpointed in .local/checkpoints/canonical/ann_tuning/. Use ANN_TUNING_FORCE=1 only when an intentional full recomputation is required.

Final ANN fitting

The canonical tuning stage retains MAX_EPOCHS = 2000. The final epoch-selection stage uses the separate FINAL_EPOCH_MAX_EPOCHS = 3000 safety ceiling. This extension was introduced after an internal, outer-assessment-free convergence diagnostic reproduced the single 2000-epoch ceiling case and reached natural patience stopping at epoch 2044 (Best_epoch = 2019). The value 3000 is a safety ceiling only; final networks are still refit for their selected Best_epoch.

This stage consumes the saved ANN_TUNING_CANONICAL_V1 selections. It never repeats hyperparameter tuning when only RUN_ANN=1 is set.

For each Scenario × outer split × ANN variant, Best_epoch is selected with early stopping using only the 800 outer-analysis individuals. The network is then reinitialized, refit on all 800 for exactly Best_epoch, and only then applied to the untouched 200 outer-assessment individuals.

final_safe_bias <- function(p,o){z<-is.finite(p)&is.finite(o);if(!any(z))return(NA_real_);mean(p[z]-o[z])}
final_safe_slope <- function(p,o){z<-is.finite(p)&is.finite(o);if(sum(z)<3L||sd(p[z])==0)return(NA_real_);as.numeric(coef(lm(o[z]~p[z]))[2])}

select_final_epoch <- function(analysis_ids,scenario_index,architecture,learning_rate,l1,l2,split_seed,fit_seed){
  iv<-make_validation_split(analysis_ids,EARLY_VALIDATION_FRACTION,split_seed)
  fi<-match(iv$train,ids);vi<-match(iv$validation,ids);if(anyNA(c(fi,vi)))stop("Final epoch-selection ID alignment failed.")
  xs<-fit_x_scaler(geno[fi,,drop=FALSE]);ys<-fit_y_scaler(phenotype[fi,scenario_index])
  xf<-scale_x(geno[fi,,drop=FALSE],xs);xv<-scale_x(geno[vi,,drop=FALSE],xs);yf<-scale_y(phenotype[fi,scenario_index],ys);yv<-scale_y(phenotype[vi,scenario_index],ys)
  set_ann_seed(fit_seed)
  xt<-torch::torch_tensor(xf,dtype=torch::torch_float())$to(device=ANN_DEVICE);yt<-torch::torch_tensor(matrix(yf,ncol=1),dtype=torch::torch_float())$to(device=ANN_DEVICE)
  xv_t<-torch::torch_tensor(xv,dtype=torch::torch_float())$to(device=ANN_DEVICE);yv_t<-torch::torch_tensor(matrix(yv,ncol=1),dtype=torch::torch_float())$to(device=ANN_DEVICE)
  model<-TorchGenomicMLP(ncol(xf),ARCHITECTURES[[architecture]])$to(device=ANN_DEVICE);opt<-torch::optim_adam(params=model$parameters,lr=learning_rate,weight_decay=0)
  best<-Inf;best_epoch<-NA_integer_;best_state<-NULL;stale<-0L;last<-0L;stopped<-FALSE;sync_device();t0<-proc.time()[["elapsed"]]
  for(epoch in seq_len(FINAL_EPOCH_MAX_EPOCHS)){last<-epoch;model$train();opt$zero_grad();pr<-model(xt);loss<-torch::nnf_mse_loss(pr,yt)+weight_penalty(model,l1,l2);loss$backward();opt$step();model$eval();vl<-torch::with_no_grad({torch::nnf_mse_loss(model(xv_t),yv_t)});v<-as.numeric(vl$to(device=torch::torch_device("cpu")));if(is.finite(v)&&v<best-MIN_DELTA){best<-v;best_epoch<-epoch;best_state<-clone_state(model);stale<-0L}else stale<-stale+1L;if(stale>=PATIENCE){stopped<-TRUE;break}}
  sync_device();secs<-proc.time()[["elapsed"]]-t0;if(is.null(best_state)||is.na(best_epoch))stop("No finite final epoch-selection solution.")
  model$load_state_dict(best_state);model$eval();pv_s<-torch::with_no_grad({as.numeric(model(xv_t)$to(device=torch::torch_device("cpu")))});pv<-unscale_y(pv_s,ys);ov<-phenotype[vi,scenario_index]
  list(best_epoch=as.integer(best_epoch),last_epoch=as.integer(last),stopped_by_patience=stopped,hit_epoch_ceiling=last>=FINAL_EPOCH_MAX_EPOCHS&&!stopped,validation_rmse=safe_rmse(pv,ov),validation_correlation=safe_cor(pv,ov),fit_seconds=secs,n_fit=length(iv$train),n_validation=length(iv$validation))
}

train_final_fixed <- function(train_ids,assessment_ids,scenario_index,architecture,learning_rate,l1,l2,epochs,seed){

  set_ann_seed(seed);xt<-torch::torch_tensor(xtr,dtype=torch::torch_float())$to(device=ANN_DEVICE);yt<-torch::torch_tensor(matrix(ytr,ncol=1),dtype=torch::torch_float())$to(device=ANN_DEVICE);xe<-torch::torch_tensor(xte,dtype=torch::torch_float())$to(device=ANN_DEVICE)
  model<-TorchGenomicMLP(ncol(xtr),ARCHITECTURES[[architecture]])$to(device=ANN_DEVICE);opt<-torch::optim_adam(params=model$parameters,lr=learning_rate,weight_decay=0);sync_device();t0<-proc.time()[["elapsed"]]
  for(epoch in seq_len(as.integer(epochs))){model$train();opt$zero_grad();pr<-model(xt);loss<-torch::nnf_mse_loss(pr,yt)+weight_penalty(model,l1,l2);loss$backward();opt$step()}

  list(prediction=unscale_y(ps,ys),fit_seconds=fit_seconds,prediction_seconds=pred_seconds,zero_variance_markers=sum(xs$zero))
}

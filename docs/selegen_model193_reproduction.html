1. Purpose

This module reproduces, in R, the numerical procedure used by SELEGEN-REML/BLUP Model 193 for the longitudinal Coffea arabica phenotypes cercosporiosis (Cer), leaf miner (BM), and production (L/P).

The objective is not to re-optimize the mixed model with a different REML implementation. Instead, the module reconstructs the iterative mixed-model equations and stopping rule that generated the audited SELEGEN outputs.

The canonical input is:

data/dados_reais/dados_selegen_cafe.txt

No derived trait-specific SELEGEN input files are required.

2. Packages

Only knitr is required for reporting. The numerical reproduction itself uses base R linear algebra.

6. Design matrices

Factor levels are ordered by first occurrence, matching the audited SELEGEN effect output.

random_fields <- c("Prog", "Parc", "Pop", "Bloc", "Int")

indicator_matrix <- function(x) {
  lev <- unique(x)

  Z <- sapply(
    lev,
    function(z) as.numeric(x == z)
  )

  if (is.null(dim(Z))) {
    Z <- matrix(Z, ncol = 1L)
  }

  colnames(Z) <- as.character(lev)

  list(
    matrix = Z,
    levels = lev
  )
}

build_design <- function(d) {
  fixed_info <- indicator_matrix(d$Fixo)
  X <- fixed_info$matrix

  random_info <- lapply(
    random_fields,
    function(field) indicator_matrix(d[[field]])
  )
  names(random_info) <- random_fields

  Zlist <- lapply(random_info, `[[`, "matrix")
  Z <- do.call(cbind, Zlist)
  W <- cbind(X, Z)

  p <- ncol(X)
  pos <- p + 1L
  slices <- list()

  for (field in random_fields) {
    q <- ncol(Zlist[[field]])
    slices[[field]] <- pos:(pos + q - 1L)
    pos <- pos + q
  }

  list(
    X = X,
    X_levels = fixed_info$levels,
    Zlist = Zlist,
    Z_levels = lapply(random_info, `[[`, "levels"),
    W = W,
    slices = slices,
    p = p
  )
}

7. Reconstructed SELEGEN update

The audited initial variance proportions are:

\[ h^2=c^2=c^2_{21}=c^2_{22}=c^2_{24}=0.10, \]

leaving a residual proportion of \(0.50\). Thus every initial variance ratio is

\[ \lambda_i=\frac{0.50}{0.10}=5. \]

At each iteration, the mixed-model equations are solved and the components are updated as

\[ \hat{\sigma}_e^2 = \frac{ e'e + \sum_i \lambda_i u_i'u_i }{ n-\operatorname{rank}(X) }, \]

and

\[ \hat{\sigma}_i^2 = \frac{ u_i'u_i + \hat{\sigma}_e^2\,\mathrm{tr}(C_{ii}) }{ q_i }. \]

selegen_iteration <- function(d, design, old_props) {
  W <- design$W
  p <- design$p
  n <- nrow(d)
  df_reml <- n - p

  lambda_old <- old_props["Ve"] / old_props[random_fields]

  penalty <- matrix(0, ncol(W), ncol(W))

  for (field in random_fields) {
    idx <- design$slices[[field]]
    penalty[idx, idx] <- diag(
      lambda_old[[field]],
      length(idx)
    )
  }

  M <- crossprod(W) + penalty
  rhs <- crossprod(W, d$y)

  coef <- solve(M, rhs)
  C <- solve(M)

  fitted <- as.numeric(W %*% coef)
  residual <- d$y - fitted

  penalized_ss <- sum(residual^2)

  for (field in random_fields) {
    idx <- design$slices[[field]]
    u <- coef[idx]

    penalized_ss <- penalized_ss +
      lambda_old[[field]] * sum(u^2)
  }

  Ve_new <- penalized_ss / df_reml

  components <- numeric(0)

  for (field in random_fields) {
    idx <- design$slices[[field]]
    u <- coef[idx]
    q <- length(idx)

    components[[field]] <- (
      sum(u^2) +
      Ve_new * sum(diag(C[idx, idx, drop = FALSE]))
    ) / q
  }

  components[["Ve"]] <- Ve_new

  Vf <- sum(components)
  new_props <- components / Vf
  lambda_new <- Ve_new / components[random_fields]

  logdet_M <- as.numeric(
    determinant(
      M,
      logarithm = TRUE
    )$modulus
  )

  q_by_effect <- vapply(
    random_fields,
    function(field) length(design$slices[[field]]),
    integer(1)
  )

  deviance <- (
    df_reml * (1 + log(Ve_new)) +
    logdet_M -
    sum(q_by_effect * log(lambda_new))
  )

  random_change <- max(
    abs(
      new_props[random_fields] -
        old_props[random_fields]
    )
  )

  beta <- coef[seq_len(p)]
  names(beta) <- as.character(design$X_levels)

  ulist <- list()

  for (field in random_fields) {
    idx <- design$slices[[field]]
    u <- coef[idx]
    names(u) <- as.character(design$Z_levels[[field]])
    ulist[[field]] <- u
  }

  list(
    components = components,
    props = new_props,
    lambda_old = lambda_old,
    lambda_new = lambda_new,
    coef = coef,
    beta = beta,
    u = ulist,
    residual = residual,
    fitted = fitted,
    C = C,
    M = M,
    logdet_M = logdet_M,
    penalized_ss = penalized_ss,
    deviance = deviance,
    random_change = random_change
  )
}

8. Iteration and stopping rule

The reconstructed stopping rule is the maximum absolute change among the five random-effect variance proportions.

fit_selegen193 <- function(trait, tolerance, max_iter = 500L) {
  d <- prepare_trait(trait)
  design <- build_design(d)

  old_props <- c(
    Prog = 0.10,
    Parc = 0.10,
    Pop = 0.10,
    Bloc = 0.10,
    Int = 0.10,
    Ve = 0.50
  )

  history <- list()
  final <- NULL

  for (iter in seq_len(max_iter)) {
    step <- selegen_iteration(
      d = d,
      design = design,
      old_props = old_props
    )

    components <- step$components
    props <- step$props

    history[[iter]] <- data.frame(
      Trait = trait,
      Iteration = iter,
      Tolerance = tolerance,
      Max_random_prop_change = step$random_change,
      Lambda_Prog_used = step$lambda_old[["Prog"]],
      Lambda_Parc_used = step$lambda_old[["Parc"]],
      Lambda_Pop_used = step$lambda_old[["Pop"]],
      Lambda_Bloc_used = step$lambda_old[["Bloc"]],
      Lambda_Int_used = step$lambda_old[["Int"]],
      Vgen_pop = components[["Prog"]],
      Vparc = components[["Parc"]],
      Vpop = components[["Pop"]],
      Vbloc = components[["Bloc"]],
      Vint = components[["Int"]],
      Ve = components[["Ve"]],
      Vf = sum(components),
      h2gen_pop = props[["Prog"]],
      c2parc = props[["Parc"]],
      h2pop = props[["Pop"]],
      c2bloc = props[["Bloc"]],
      c2int = props[["Int"]],
      residual_prop = props[["Ve"]],
      LogDet_M = step$logdet_M,
      Penalized_SS = step$penalized_ss,
      Deviance = step$deviance,
      stringsAsFactors = FALSE
    )

    if (step$random_change <= tolerance) {
      final <- step
      break
    }

    old_props <- props
  }

  if (is.null(final)) {
    stop(
      trait,
      ": Model 193 reconstruction did not meet the stopping rule within ",
      max_iter,
      " iterations."
    )
  }

  history_df <- do.call(rbind, history)
  rownames(history_df) <- NULL

  list(
    trait = trait,
    data = d,
    design = design,
    final = final,
    history = history_df,
    iterations = nrow(history_df),
    tolerance = tolerance
  )
}

safe_cor <- function(pred, obs) {
  ok <- is.finite(pred) & is.finite(obs)
  if (sum(ok) < 3L || sd(pred[ok]) == 0 || sd(obs[ok]) == 0) return(NA_real_)
  cor(pred[ok], obs[ok])
}

safe_rmse <- function(pred, obs) {
  ok <- is.finite(pred) & is.finite(obs)
  if (!any(ok)) return(NA_real_)
  sqrt(mean((pred[ok] - obs[ok])^2))
}

safe_mae <- function(pred, obs) {
  ok <- is.finite(pred) & is.finite(obs)
  if (!any(ok)) return(NA_real_)
  mean(abs(pred[ok] - obs[ok]))
}

safe_bias <- function(pred, obs) {
  ok <- is.finite(pred) & is.finite(obs)
  if (!any(ok)) return(NA_real_)
  mean(pred[ok] - obs[ok])
}

safe_slope <- function(pred, obs) {
  ok <- is.finite(pred) & is.finite(obs)
  if (sum(ok) < 3L || sd(pred[ok]) == 0) return(NA_real_)
  as.numeric(coef(lm(obs[ok] ~ pred[ok]))[2])
}

safe_r2 <- function(pred, obs) {
  ok <- is.finite(pred) & is.finite(obs)
  if (sum(ok) < 2L) return(NA_real_)
  den <- sum((obs[ok] - mean(obs[ok]))^2)
  if (den == 0) return(NA_real_)
  1 - sum((obs[ok] - pred[ok])^2) / den
}

metrics <- function(pred, obs, prefix) {
  z <- c(
    safe_cor(pred, obs),
    safe_rmse(pred, obs),
    safe_mae(pred, obs),
    safe_bias(pred, obs),
    safe_slope(pred, obs),
    safe_r2(pred, obs)
  )
  names(z) <- paste0(
    prefix,
    c("_Correlation", "_RMSE", "_MAE", "_Bias", "_Slope", "_R2")
  )
  z
}

extract_u <- function(fit, dat, factor_name, exact_name) {
  unames <- names(fit$uList)
  if (exact_name %in% unames) {
    nm <- exact_name
  } else {
    token <- paste0("ism(", factor_name, ")")
    hit <- grep(token, unames, fixed = TRUE)
    if (length(hit) != 1L) {
      stop(
        paste(
          "Could not identify random component for", factor_name,
          ". Available:", paste(unames, collapse = " | ")
        )
      )
    }
    nm <- unames[hit]
  }

  u <- fit$uList[[nm]]
  if (is.data.frame(u)) u <- as.matrix(u)
  if (is.vector(u)) u <- matrix(u, ncol = 1L)

  X <- model.matrix(
    as.formula(paste0("~", factor_name, " - 1")),
    data = dat
  )

  if (ncol(X) != nrow(u)) {
    stop(
      paste(
        "Random-effect dimension mismatch for", factor_name,
        ":", ncol(X), "vs", nrow(u)
      )
    )
  }

  as.numeric(X %*% u[, 1L, drop = FALSE])
}

Fit GBLUP-ADE

The A, D, and E relationship matrices are built once from the complete genotype matrix and reused across tasks. Assessment phenotypes are set to NA before every fit. Genotypes of assessment individuals are allowed in the relationship matrices; their phenotypes and true genetic values are not used for fitting.
